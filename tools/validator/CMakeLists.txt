cmake_minimum_required(VERSION 3.27)
# Load the cross toolchain before the SDK detects the target platform.
project(WaveFM1Validator LANGUAGES C CXX)
set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
set(SMTG_RUN_VST_VALIDATOR OFF CACHE BOOL "" FORCE)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(VST3_SDK_ROOT "${ROOT}/build/cross-windows/cpm/vst3sdk" CACHE PATH "Downloaded official VST3 SDK")
add_subdirectory("${VST3_SDK_ROOT}" sdk)
if(MINGW)
    target_link_options(validator PRIVATE -static -static-libgcc -static-libstdc++ -municode)
    target_link_libraries(validator PRIVATE shell32 ole32 uuid)
endif()
