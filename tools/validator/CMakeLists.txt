cmake_minimum_required(VERSION 3.27)
# Load the cross toolchain before the SDK detects the target platform.
project(WaveFM1Validator LANGUAGES C CXX)
set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
set(SMTG_RUN_VST_VALIDATOR OFF CACHE BOOL "" FORCE)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(VST3_SDK_ROOT "${ROOT}/build/cross-windows/cpm/vst3sdk" CACHE PATH "Downloaded official VST3 SDK")
add_subdirectory("${VST3_SDK_ROOT}" sdk)
set(empty_latency_sources
    empty-latency.cpp
    "${VST3_SDK_ROOT}/public.sdk/source/vst/hosting/plugprovider.cpp"
    "${VST3_SDK_ROOT}/public.sdk/samples/vst-hosting/validator/source/usediids.cpp"
)
if(APPLE)
    list(APPEND empty_latency_sources "${VST3_SDK_ROOT}/public.sdk/source/vst/hosting/module_mac.mm")
    set_source_files_properties("${VST3_SDK_ROOT}/public.sdk/source/vst/hosting/module_mac.mm"
        PROPERTIES COMPILE_FLAGS "-fobjc-arc")
elseif(WIN32)
    list(APPEND empty_latency_sources "${VST3_SDK_ROOT}/public.sdk/source/vst/hosting/module_win32.cpp")
else()
    list(APPEND empty_latency_sources "${VST3_SDK_ROOT}/public.sdk/source/vst/hosting/module_linux.cpp")
endif()
add_executable(fm1-latency-check ${empty_latency_sources})
target_compile_features(fm1-latency-check PRIVATE cxx_std_17)
target_link_libraries(fm1-latency-check PRIVATE sdk_hosting)
if(APPLE)
    find_library(COCOA_FRAMEWORK Cocoa REQUIRED)
    target_link_libraries(fm1-latency-check PRIVATE ${COCOA_FRAMEWORK})
endif()
if(MINGW)
    target_link_options(validator PRIVATE -static -static-libgcc -static-libstdc++ -municode)
    target_link_libraries(validator PRIVATE shell32 ole32 uuid)
    target_link_options(fm1-latency-check PRIVATE -static -static-libgcc -static-libstdc++)
    target_link_libraries(fm1-latency-check PRIVATE shell32 ole32 uuid)
endif()
